        // --- Theme Toggle Logic ---
        const htmlElement = document.documentElement;

        function updateTheme(theme) {
            const themeToggleBtn = document.getElementById('theme-toggle');
            const label = document.getElementById('current-theme-label');

            if (theme === 'dark') {
                htmlElement.classList.add('dark');
                localStorage.setItem('theme', 'dark');
            } else {
                htmlElement.classList.remove('dark');
                localStorage.setItem('theme', 'light');
            }
            
            // Update the icon
            if (themeToggleBtn) {
                const icon = themeToggleBtn.querySelector('i');
                if (icon) {
                    icon.setAttribute('data-lucide', theme === 'dark' ? 'sun' : 'moon');
                    lucide.createIcons();
                }
            }

            // Update the text label
            if (label) {
                label.textContent = theme === 'dark' ? 'Dark Mode' : 'Light Mode';
            }
        }

        window.toggleTheme = function() {
            const currentTheme = localStorage.getItem('theme') === 'dark' ? 'dark' : 'light';
            updateTheme(currentTheme === 'dark' ? 'light' : 'dark');
        }

        // Initialize theme on load
        function initTheme() {
            const savedTheme = localStorage.getItem('theme');
            const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
            
            if (savedTheme) {
                updateTheme(savedTheme);
            } else if (prefersDark) {
                updateTheme('dark');
            } else {
                updateTheme('light');
            }
        }

        window.copyToClipboard = function(text, buttonElement) {
            const textarea = document.createElement('textarea');

            document.body.appendChild(textarea);
            textarea.select();

            try {

        // --- Firebase Setup and Authentication ---
        async function setupFirebase() {
            try {
                app = initializeApp(firebaseConfig);
                db = getFirestore(app);
                auth = getAuth(app);

                onAuthStateChanged(auth, (user) => {
                    if (user) {
                        currentUserId = user.uid;
                        document.getElementById('user-id-display').textContent = `User: ${currentUserId.substring(0, 8)}...`;

                        // Start listening for all data streams
                        loadPosts();
                        loadMainArticles(); 
                    } else {
                        currentUserId = 'anonymous-' + crypto.randomUUID().substring(0, 8);
                        document.getElementById('user-id-display').textContent = `User: ${currentUserId}`;
                        signInAnonymously(auth).catch((error) => console.error("Anonymous Sign-in failed:", error));
                    }
                });

                if (initialAuthToken) {
                    await signInWithCustomToken(auth, initialAuthToken);
                }
            } catch (error) {
                console.error("Firebase Initialization Error:", error);
            }
        }
        
        // --- Dynamic Content Logic (Read-Only) ---
        function loadMainArticles() {
            if (!db) return;

            const articlesPath = `/artifacts/${appId}/public/data/main_articles`;
            const articlesCollection = collection(db, articlesPath);
            
            // Listen for all articles in the collection
            const q = query(articlesCollection);

            onSnapshot(q, (snapshot) => {
                const allArticles = [];
                
                snapshot.forEach((doc) => {
                    const article = { ...doc.data(), id: doc.id };
                    allArticles.push(article);
                });

                renderMainArticles(allArticles);
            }, (error) => {
                console.error("Error loading main articles:", error);
            });
        }
        
        function renderMainArticles(articles) {
            // 1. Find Hero Article (limit to one 'hero' section article)
            const heroArticle = articles.find(a => a.section === 'hero');
            const heroElement = document.getElementById('hero-story-container');
            heroElement.innerHTML = ''; // Clear existing content

            if (heroArticle) {
                const articleHtml = `
                    <img class="w-full h-56 md:h-96 object-cover" 
                         src="${heroArticle.image_url || window.getPlaceholderUrl(1200, 600, heroArticle.category.toUpperCase())}"

                         onerror="this.onerror=null;this.src=window.getPlaceholderUrl(1200, 600, '${heroArticle.category.toUpperCase()}');">
                    <div class="p-4 md:p-6">
                        <span class="inline-block bg-brand-red text-white text-xs font-semibold px-3 py-1 rounded-full mb-2 uppercase shadow-md">${heroArticle.category}</span>
                        <h2 class="text-2xl md:text-3xl font-extrabold text-gray-900 leading-tight mb-3 dark:text-gray-100">
                            ${heroArticle.title}
                        </h2>
                        <p class="text-gray-600 mb-4 dark:text-gray-300">
                            ${heroArticle.summary}
                        </p>
                        <a href="#" class="text-brand-red font-bold hover:underline transition duration-150 flex items-center">
                            Read Full Story 
                            <i data-lucide="arrow-right" class="w-4 h-4 ml-1"></i>
                        </a>
                    </div>
                `;
                heroElement.innerHTML = articleHtml;
                lucide.createIcons(); // Re-render lucide icons in new HTML
            } else {
                 heroElement.innerHTML = `<div class="p-6 text-center text-gray-500 dark:text-gray-400">No featured hero story published yet.</div>`;
            }

            // 2. Find Finance Articles (limit to 3 'finance' section articles)
            const financeArticles = articles
                .filter(a => a.section === 'finance')
                .sort((a, b) => b.timestamp?.seconds - a.timestamp?.seconds)
                .slice(0, 3);

            const financeListElement = document.getElementById('finance-list');
            financeListElement.innerHTML = '';

            if (financeArticles.length > 0) {
                financeArticles.forEach(article => {
                    const dateDisplay = article.timestamp ? new Date(article.timestamp.seconds * 1000).toLocaleDateString() : 'Just now';
                    financeListElement.innerHTML += `
                        <li class="p-3 bg-white rounded-lg shadow-sm hover:shadow-md transition duration-200 border-l-4 border-highlight-orange dark:bg-gray-800">
                            <a href="#" class="text-gray-800 font-semibold hover:text-brand-red cursor-pointer block dark:text-gray-200">
                                ${article.title}
                            </a>
                            <p class="text-xs text-gray-500 mt-1 dark:text-gray-400">${article.category} | ${dateDisplay}</p>
                        </li>
                    `;
                });
            } else {
                financeListElement.innerHTML = `<li class="p-3 text-gray-500 dark:text-gray-400 text-sm">No finance articles published.</li>`;
            }
        }
        
        // --- Discussion Board Logic ---
        function loadPosts() {
            if (!db) return;

            const discussionPath = `/artifacts/${appId}/public/data/discussion_posts`;
            const postsCollection = collection(db, discussionPath);
            
            // Query: Get posts and order by timestamp (descending for latest first)
            const q = query(postsCollection, orderBy('timestamp', 'desc'), limit(20)); // Limit to last 20 posts

            onSnapshot(q, (snapshot) => {
                const feedElement = document.getElementById('discussion-feed');
                if (!feedElement) return;

                const posts = [];
                snapshot.forEach((doc) => posts.push(doc.data()));
                
                // Clear existing feed
                feedElement.innerHTML = '';
                
                if (posts.length === 0) {
                    feedElement.innerHTML = '<p class="text-gray-500 dark:text-gray-400 text-sm italic p-4 text-center">Be the first to comment!</p>';
                    return;
                }

                // Reverse the array to display chronologically in the box (oldest on top, newest at bottom)
                posts.reverse().forEach((post) => {
                    const postElement = createPostElement(post);
                    feedElement.appendChild(postElement);
                });

                // Scroll to the newest post (bottom of the feed)
                feedElement.scrollTop = feedElement.scrollHeight;
            }, (error) => {
                console.error("Error listening to discussion posts:", error);
            });
        }

        function createPostElement(post) {
            const date = post.timestamp ? new Date(post.timestamp.seconds * 1000).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }) : 'Now';
            const userDisplay = post.userId ? post.userId.substring(0, 8) + '...' : 'Unknown';

            const div = document.createElement('div');
            div.className = 'p-3 bg-white border-b border-gray-200 last:border-b-0 rounded-lg shadow-sm dark:bg-gray-700 dark:border-gray-600';
            div.innerHTML = `
                <p class="text-gray-800 text-sm break-words dark:text-gray-200">${post.content}</p>
                <p class="text-xs mt-1 text-gray-500 dark:text-gray-400">
                    <span class="font-bold text-brand-red">${userDisplay}</span> 
                    <span class="ml-2">${date}</span>
                </p>
            `;
            return div;
        }

        window.submitPost = async function (event) {
            event.preventDefault();
            
            const contentInput = document.getElementById('post-content');
            const content = contentInput.value.trim();

            if (!content || !auth.currentUser || !db) return;
            
            const submitButton = document.getElementById('submit-post-btn');
            submitButton.disabled = true;
            submitButton.textContent = 'Posting...';

            try {
                const discussionPath = `/artifacts/${appId}/public/data/discussion_posts`;
                const postsCollection = collection(db, discussionPath);
                
                await addDoc(postsCollection, {
                    userId: auth.currentUser.uid,
                    content: content,
                    timestamp: serverTimestamp()
                });

                contentInput.value = ''; // Clear input on success
            } catch (error) {
                console.error("Error submitting post:", error);
            } finally {
                submitButton.disabled = false;
                submitButton.textContent = 'Post Comment';
            }
        }
